<script>
    let peer = null;
    let conn = null;
    let originalImage = null;
    const urlParams = new URLSearchParams(window.location.search);
    const targetPeerId = urlParams.get('connect');

    function initPeer() {
        peer = new Peer();

        peer.on('open', (id) => {
            if (targetPeerId) {
                // If scanned from mobile
                document.getElementById('connectionStatus').innerText = "પીસી સાથે જોડાઈ રહ્યું છે...";
                conn = peer.connect(targetPeerId);
                setupConnection();
            } else {
                // If host (PC)
                const currentUrl = window.location.origin + window.location.pathname + '?connect=' + id;
                document.getElementById('qrcode').innerHTML = "";
                new QRCode(document.getElementById("qrcode"), {
                    text: currentUrl,
                    width: 170,
                    height: 170
                });
                document.getElementById('connectionStatus').innerText = "QR સ્કેન થવાની રાહ જોવાઈ રહી છે...";
            }
        });

        peer.on('connection', (c) => {
            conn = c;
            setupConnection();
        });
    }

    function setupConnection() {
        conn.on('open', () => {
            const statusEl = document.getElementById('connectionStatus');
            statusEl.className = "status-pill status-connected";
            statusEl.innerText = "સફળતાપૂર્વક જોડાઈ ગયું! (Connected)";
        });

        conn.on('data', (data) => {
            if (data.file) {
                const blob = new Blob([data.file], { type: data.fileType });
                const url = URL.createObjectURL(blob);
                const list = document.getElementById('receivedFiles');
                const li = document.createElement('li');
                li.innerHTML = `<span>📄 <b>${data.fileName}</b></span> <a href="${url}" download="${data.fileName}">ડાઉનલોડ ⬇</a>`;
                list.prepend(li);
            }
        });
    }

    function sendFile() {
        const fileInput = document.getElementById('fileInput');
        if (!fileInput.files.length) return alert('કૃપા કરીને ફાઇલ પસંદ કરો');
        if (!conn || !conn.open) return alert('પહેલાં બંને ડિવાઇસ કનેક્ટ કરો (QR સ્કેન કરો)');

        const file = fileInput.files[0];
        conn.send({ file: file, fileName: file.name, fileType: file.type });
        alert(`'${file.name}' મોકલી દેવામાં આવી છે!`);
        fileInput.value = '';
    }

    function loadPhoto(e) {
        const file = e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(event) {
            originalImage = new Image();
            originalImage.onload = function() {
                document.getElementById('previewCanvas').style.display = 'block';
                applyFilters();
            }
            originalImage.src = event.target.result;
        }
        reader.readAsDataURL(file);
    }

    function applyFilters() {
        if (!originalImage) return;
        const canvas = document.getElementById('previewCanvas');
        const ctx = canvas.getContext('2d');

        canvas.width = originalImage.width;
        canvas.height = originalImage.height;

        const sharp = document.getElementById('sharpness').value;
        const cont = document.getElementById('contrast').value;
        const bright = document.getElementById('brightness').value;

        document.getElementById('valSharp').innerText = sharp;
        document.getElementById('valContrast').innerText = cont;
        document.getElementById('valBright').innerText = bright;

        // Apply visual CSS filters onto the canvas context
        ctx.filter = `contrast(${cont}) brightness(${bright})`;
        ctx.drawImage(originalImage, 0, 0);

        // Simple convolution matrix for sharpness/unblur
        if (parseFloat(sharp) > 1) {
            applySharpness(ctx, canvas.width, canvas.height, parseFloat(sharp));
        }
    }

    function applySharpness(ctx, w, h, strength) {
        const imgData = ctx.getImageData(0, 0, w, h);
        const data = imgData.data;
        const copy = new Uint8ClampedArray(data);
        const factor = (strength - 1) * 0.5;

        for (let y = 1; y < h - 1; y++) {
            for (let x = 1; x < w - 1; x++) {
                const idx = (y * w + x) * 4;
                for (let c = 0; c < 3; c++) {
                    const up = ((y - 1) * w + x) * 4 + c;
                    const down = ((y + 1) * w + x) * 4 + c;
                    const left = (y * w + (x - 1)) * 4 + c;
                    const right = (y * w + (x + 1)) * 4 + c;
                    const center = idx + c;

                    const laplacian = (copy[center] * 5) - (copy[up] + copy[down] + copy[left] + copy[right]);
                    data[center] = Math.min(255, Math.max(0, copy[center] + laplacian * factor));
                }
            }
        }
        ctx.putImageData(imgData, 0, 0);
    }

    function sendEnhancedPhoto() {
        if (!conn || !conn.open) return alert('પહેલાં બંને ડિવાઇસ કનેક્ટ કરો');
        const canvas = document.getElementById('previewCanvas');
        if (canvas.style.display === 'none') return alert('ફોટો સિલેક્ટ કરો');

        canvas.toBlob((blob) => {
            conn.send({ file: blob, fileName: 'enhanced_photo_' + Date.now() + '.jpg', fileType: 'image/jpeg' });
            alert('સુધારેલો ફોટો સેન્ડ થઈ ગયો છે!');
        }, 'image/jpeg', 0.95);
    }

    window.onload = initPeer;
</script>
